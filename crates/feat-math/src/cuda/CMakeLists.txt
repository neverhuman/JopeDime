cmake_minimum_required(VERSION 3.24)
project(standalone_model LANGUAGES CXX)
option(JOPE_CUDA_HOST_ONLY "Build the complete host API without a CUDA toolkit" OFF)
find_program(RUSTC rustc REQUIRED)
set(NATIVE "${CMAKE_CURRENT_BINARY_DIR}/libmodel_native.a")
add_custom_command(OUTPUT "${NATIVE}"
  COMMAND "${RUSTC}" --edition=2021 -O -C panic=unwind
    --crate-type=staticlib "${CMAKE_CURRENT_SOURCE_DIR}/model_native.rs" -o "${NATIVE}"
  DEPENDS model_native.rs model_core.rs VERBATIM)
add_custom_target(model_native_build DEPENDS "${NATIVE}")
add_library(model_native STATIC IMPORTED)
set_target_properties(model_native PROPERTIES IMPORTED_LOCATION "${NATIVE}")
add_dependencies(model_native model_native_build)
if(JOPE_CUDA_HOST_ONLY)
  set_source_files_properties(model.cu PROPERTIES LANGUAGE CXX COMPILE_FLAGS "-x c++")
else()
  enable_language(CUDA)
  find_package(CUDAToolkit REQUIRED)
endif()
add_library(model STATIC model.cu)
target_include_directories(model PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}")
target_compile_features(model PUBLIC cxx_std_17)
set_target_properties(model PROPERTIES CUDA_STANDARD 17 CUDA_STANDARD_REQUIRED ON)
target_link_libraries(model PUBLIC model_native ${CMAKE_DL_LIBS} pthread m)
if(JOPE_CUDA_HOST_ONLY)
  target_compile_definitions(model PRIVATE JOPE_CUDA_HOST_ONLY)
  target_compile_options(model PRIVATE -ffp-contract=off -fno-fast-math)
else()
  target_compile_options(model PRIVATE --fmad=false --ftz=false --prec-div=true --prec-sqrt=true
    -Xcompiler=-ffp-contract=off -Xcompiler=-fno-fast-math)
  target_link_libraries(model PUBLIC CUDA::cudart)
endif()
