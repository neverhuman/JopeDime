cmake_minimum_required(VERSION 3.24)
project(standalone_model LANGUAGES CXX)
set(JOPE_MODEL_TARGET "__CMAKE_TARGET__")
# Identical bundles share one target; distinct bundles have source-derived names.
if(TARGET "${JOPE_MODEL_TARGET}")
  return()
endif()
option(JOPE_CUDA_HOST_ONLY "Build the complete host API without a CUDA toolkit" OFF)
find_program(RUSTC rustc REQUIRED)
set(NATIVE "${CMAKE_CURRENT_BINARY_DIR}/libmodel_native.a")
add_custom_command(OUTPUT "${NATIVE}"
  COMMAND "${RUSTC}" --edition=2021 -O -C panic=unwind
    --crate-type=staticlib "${CMAKE_CURRENT_SOURCE_DIR}/model_native.rs" -o "${NATIVE}"
  DEPENDS model_native.rs model_core.rs VERBATIM)
add_custom_target("${JOPE_MODEL_TARGET}_native_build" DEPENDS "${NATIVE}")
add_library("${JOPE_MODEL_TARGET}_native" STATIC IMPORTED GLOBAL)
set_target_properties("${JOPE_MODEL_TARGET}_native" PROPERTIES IMPORTED_LOCATION "${NATIVE}")
add_dependencies("${JOPE_MODEL_TARGET}_native" "${JOPE_MODEL_TARGET}_native_build")
if(JOPE_CUDA_HOST_ONLY)
  set_source_files_properties(model.cu PROPERTIES LANGUAGE CXX COMPILE_FLAGS "-x c++")
else()
  enable_language(CUDA)
  find_package(CUDAToolkit REQUIRED)
endif()
add_library("${JOPE_MODEL_TARGET}" STATIC model.cu)
if(NOT TARGET model)
  add_library(model ALIAS "${JOPE_MODEL_TARGET}")
endif()
target_include_directories("${JOPE_MODEL_TARGET}" PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}")
target_compile_features("${JOPE_MODEL_TARGET}" PUBLIC cxx_std_17)
set_target_properties("${JOPE_MODEL_TARGET}" PROPERTIES CUDA_STANDARD 17 CUDA_STANDARD_REQUIRED ON)
target_link_libraries("${JOPE_MODEL_TARGET}" PUBLIC "${JOPE_MODEL_TARGET}_native" ${CMAKE_DL_LIBS} pthread m)
if(JOPE_CUDA_HOST_ONLY)
  target_compile_definitions("${JOPE_MODEL_TARGET}" PRIVATE JOPE_CUDA_HOST_ONLY)
  target_compile_options("${JOPE_MODEL_TARGET}" PRIVATE -ffp-contract=off -fno-fast-math)
else()
  target_compile_options("${JOPE_MODEL_TARGET}" PRIVATE --fmad=false --ftz=false --prec-div=true --prec-sqrt=true
    -Xcompiler=-ffp-contract=off -Xcompiler=-fno-fast-math)
  target_link_libraries("${JOPE_MODEL_TARGET}" PUBLIC CUDA::cudart)
endif()
