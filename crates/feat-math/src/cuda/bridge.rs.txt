// Extra bounded native operations. All caller output commits after successful work.
fn retained(handle:u64)->Result<Arc<model_core::CudaModel>,i32>{
    handles().lock().map_err(|_|INTERNAL_ERROR)?.models.get(&handle).cloned().ok_or(INVALID_HANDLE)
}
#[no_mangle]
pub unsafe extern "C" fn jope_cuda_native_open(out:*mut u64)->i32{
    boundary(||{
        if !buffer(out,1){return Err(INVALID_ARGUMENT);}
        let model=Arc::new(model_core::CudaModel::open()?);
        let mut table=handles().lock().map_err(|_|INTERNAL_ERROR)?;
        let id=table.next;table.next=id.checked_add(1).ok_or(INTERNAL_ERROR)?;
        table.models.insert(id,model);unsafe{out.write(id);}Ok(())
    })
}
#[no_mangle]
pub unsafe extern "C" fn jope_cuda_native_linear_width(handle:u64,out:*mut usize)->i32{
    boundary(||{
        if !buffer(out,1){return Err(INVALID_ARGUMENT);}
        let width=retained(handle)?.linear_width()?;
        unsafe{out.write(width);}Ok(())
    })
}
#[no_mangle]
pub unsafe extern "C" fn jope_cuda_native_linear(
    handle:u64,x:*const f64,x_len:usize,rows:usize,cols:usize,
    values:*mut f64,values_len:usize,weights:*mut f64,weights_len:usize,
)->i32{
    boundary(||{
        if !shape(rows,cols,x_len)||!buffer(x,x_len)||!buffer(values,values_len)||!buffer(weights,weights_len){return Err(INVALID_ARGUMENT);}
        let model=retained(handle)?;
        if cols!=model.width{return Err(INVALID_ARGUMENT);}
        let p=model.linear_width()?;
        if rows.checked_mul(p)!=Some(values_len)||p.checked_add(1)!=Some(weights_len){return Err(INVALID_ARGUMENT);}
        // Two mutable output ranges must be disjoint; input may alias either.
        let a=values as usize;let b=weights as usize;
        let a_end=a.checked_add(values_len*8).ok_or(INVALID_ARGUMENT)?;
        let b_end=b.checked_add(weights_len*8).ok_or(INVALID_ARGUMENT)?;
        if values_len!=0&&weights_len!=0&&a<b_end&&b<a_end{return Err(INVALID_ARGUMENT);}
        let x=unsafe{rows_from(x,rows,cols)};
        let (v,w)=model.linear(&x)?;
        if v.len()!=values_len||w.len()!=weights_len{return Err(INTERNAL_ERROR);}
        unsafe{
            if values_len!=0{std::ptr::copy_nonoverlapping(v.as_ptr(),values,values_len);}
            std::ptr::copy_nonoverlapping(w.as_ptr(),weights,weights_len);
        }Ok(())
    })
}
#[no_mangle]
pub unsafe extern "C" fn jope_cuda_native_text(
    handle:u64,text:*const u8,text_len:usize,out:*mut f64,out_len:usize,
)->i32{
    boundary(||{
        if !buffer(text,text_len)||!buffer(out,out_len){return Err(INVALID_ARGUMENT);}
        let bytes=if text_len==0{Vec::new()}else{unsafe{std::slice::from_raw_parts(text,text_len)}.to_vec()};
        let csv=std::str::from_utf8(&bytes).map_err(|_|INVALID_ARGUMENT)?;
        let predictions=retained(handle)?.predict_text(csv)?;
        if predictions.len()!=out_len{return Err(INVALID_ARGUMENT);}
        if out_len!=0{unsafe{std::ptr::copy_nonoverlapping(predictions.as_ptr(),out,out_len);}}
        Ok(())
    })
}
