
// Private, instance-local transfer storage. Addresses are produced by the
// allocator and are never accepted as arguments to a Rust function.
thread_local! {
    static GO_VALUES: std::cell::RefCell<Vec<f64>> = const { std::cell::RefCell::new(Vec::new()) };
    static GO_LABELS: std::cell::RefCell<Vec<f64>> = const { std::cell::RefCell::new(Vec::new()) };
    static GO_OUTPUT: std::cell::RefCell<Vec<f64>> = const { std::cell::RefCell::new(Vec::new()) };
    static GO_TEXT: std::cell::RefCell<Vec<u8>> = const { std::cell::RefCell::new(Vec::new()) };
}
const GO_MAX_ELEMENTS: usize = 16_777_216;
#[no_mangle]
pub extern "C" fn model_abi_version() -> u32 { 1 }
#[no_mangle]
pub extern "C" fn model_values(len: usize) -> usize {
    if len > GO_MAX_ELEMENTS { return 0; }
    GO_VALUES.with(|b| { let mut b=b.borrow_mut(); b.resize(len,0.0); b.as_mut_ptr() as usize })
}
#[no_mangle]
pub extern "C" fn model_labels(len: usize) -> usize {
    if len > GO_MAX_ELEMENTS { return 0; }
    GO_LABELS.with(|b| { let mut b=b.borrow_mut(); b.resize(len,0.0); b.as_mut_ptr() as usize })
}
#[no_mangle]
pub extern "C" fn model_text(len: usize) -> usize {
    if len > GO_MAX_ELEMENTS { return 0; }
    GO_TEXT.with(|b| { let mut b=b.borrow_mut(); b.resize(len,0); b.as_mut_ptr() as usize })
}
#[no_mangle]
pub extern "C" fn model_output() -> usize {
    GO_OUTPUT.with(|b| b.borrow().as_ptr() as usize)
}
#[no_mangle]
pub extern "C" fn model_output_len() -> usize {
    GO_OUTPUT.with(|b| b.borrow().len())
}
fn go_rows(rows: usize, cols: usize) -> Option<Vec<Vec<f64>>> {
    if cols == 0 || rows.checked_mul(cols)? > GO_MAX_ELEMENTS { return None; }
    GO_VALUES.with(|b| {
        let b=b.borrow();
        if b.len() != rows*cols { return None; }
        Some(b.chunks_exact(cols).map(|row| row.to_vec()).collect())
    })
}
fn go_clear_buffers() {
    GO_VALUES.with(|b| *b.borrow_mut()=Vec::new());
    GO_LABELS.with(|b| *b.borrow_mut()=Vec::new());
    GO_OUTPUT.with(|b| *b.borrow_mut()=Vec::new());
    GO_TEXT.with(|b| *b.borrow_mut()=Vec::new());
}
