
fn raw_json_string(value: &str) -> String {
    let mut output=String::from("\"");
    for character in value.chars() {
        match character {
            '"'=>output.push_str("\\\""), '\\'=>output.push_str("\\\\"),
            '\u{8}'=>output.push_str("\\b"), '\u{c}'=>output.push_str("\\f"),
            '\n'=>output.push_str("\\n"), '\r'=>output.push_str("\\r"), '\t'=>output.push_str("\\t"),
            value if value<'\u{20}'=>output.push_str(&format!("\\u{:04x}",value as u32)),
            value=>output.push(value),
        }
    }
    output.push('"'); output
}
/// Predict original string/null features in schema order. Returns the complete JSON prediction contract.
pub fn predict_raw(rows: &[Vec<Option<String>>]) -> Result<String,String> {
    let kinds: &[bool]=&[__KINDS__];
    let mut encoded=Vec::with_capacity(rows.len());
    for row in rows {
        if row.len()!=kinds.len() { return Err("input shape differs".into()); }
        let mut output=Vec::new();
        for (value,categorical) in row.iter().zip(kinds) {
            output.push(match value {
                None=>String::new(),
                Some(value) if *categorical=>format!("jope.category.v1:{}",raw_json_string(value)),
                Some(value)=>{
                    let number=value.trim().parse::<f64>().map_err(|_|"invalid numeric input".to_string())?;
                    if !number.is_finite() { return Err("numeric features must be finite or null".into()); }
                    value.clone()
                }
            });
        }
        encoded.push(output);
    }
    __COMPUTE__
    let width=__WIDTH__;
    if values.len()!=rows.len() || values.iter().any(|row| row.len()!=width) { return Err("fitted output shape differs".into()); }
    if !__REGRESSION__ && values.iter().any(|row| row.iter().any(|v| !v.is_finite() || *v<0.0 || *v>1.0) || (row.iter().sum::<f64>()-1.0).abs()>1e-6) { return Err("invalid fitted probability matrix".into()); }
    let mut output=String::from(__PREFIX__);
    for (index,row) in values.iter().enumerate() {
        if index>0 { output.push(','); }
        if row.iter().any(|value| !value.is_finite()) { return Err("nonfinite fitted output".into()); }
        if __REGRESSION__ { output.push_str(&format!("{:?}",row[0])); }
        else {
            output.push('[');
            for (column,value) in row.iter().enumerate() {
                if column>0 { output.push(','); }
                output.push_str(&format!("{value:?}"));
            }
            output.push(']');
        }
    }
    output.push_str(__SUFFIX__); Ok(output)
}
