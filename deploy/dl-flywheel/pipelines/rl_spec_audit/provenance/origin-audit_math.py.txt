def softmax(z):
 e=[math.exp(x-max(z)) for x in z];return [x/sum(e) for x in e]
def grad(p,a):return [float(k==a)-p[k] for k in range(len(p))]
def enumerated(p,q,h,g,*,mu=None,weight='none',average=False,dense=False):
 mu=mu or p;n=len(p);out=[0.]*n;value=0.
 for a in itertools.product(range(n),repeat=g):
  mass=math.prod(mu[i]for i in a);values=[q[i]for i in a];reward=max([h,*values])-h;value+=mass*reward
  for j,action in enumerate(a):
   others=values[:j]+values[j+1:]
   adv=values[j]-sum(others)/len(others)if dense else max(0.,values[j]-max([h,*others]))
   factor=math.prod(p[k]/mu[k]for k in a)if weight=='joint'else p[action]/mu[action]if weight=='item'else 1.
   for k,v in enumerate(grad(p,action)):out[k]+=mass*factor*adv*v/(g if average else 1.)
 return value,out

def expected(z,q,h,g):return enumerated(softmax(z),q,h,g)[0]
def finite(z,q,h,g,eps=1e-5):
 out=[]

  a=z[:];b=z[:];a[i]+=eps;b[i]-=eps;out.append((expected(a,q,h,g)-expected(b,q,h,g))/(2*eps))
 return out
def error(a,b):return max(abs(x-y)for x,y in zip(a,b))
