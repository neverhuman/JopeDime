% A \FloatBarrier that flushes figure* floats onto a float-only page can leave
% the next column with the previous page's reduced text room, so that column
% stops after a few lines (the Section 7 opening after the Section 6.8 figures).
% When the barrier leaves an empty column holding no column floats, give it the
% full column height again.
% It also breaks the column first (\penalty0): otherwise, when the paragraph
% before the barrier spills into the next column while floats are held,
% placeins' \newpage lands after the spilled line and ejects that column too.
\makeatletter
\let\fb@plainbarrier\FloatBarrier
\renewcommand{\FloatBarrier}{\par\penalty\z@\fb@plainbarrier\fb@freshcolumn}
\newcommand{\fb@freshcolumn}{%
  \ifdim\pagegoal=\maxdimen
    \ifx\@toplist\@empty\ifx\@botlist\@empty
      \ifdim\@colroom<\@colht
        \global\@colroom\@colht\global\vsize\@colht
      \fi
    \fi\fi
  \fi}
\makeatother

  \begingroup\let\@elt\relax
  \edef\@tempa{\@botlist\@deferlist\@dbldeferlist}%
  \ifx\@tempa\@empty\global\fb@heldfalse\else\global\fb@heldtrue\fi
  \endgroup
  \iffb@held
    \global\advance\fb@flushcount\@ne
    \ifnum\fb@flushcount>12 \clearpage\else\ifdim\pagegoal=\maxdimen\null\fi\newpage\fi
    \expandafter\fb@flushstep
  \fi}
