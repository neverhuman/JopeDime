struct OrderedRow {OrderedGroup group;double timestamp;std::vector<double> values;};
struct IntegrationFittedModel {
 IntegrationState state;unsigned task;
 std::pair<std::vector<double>,std::vector<std::vector<std::uint32_t>>> predict(const std::vector<OrderedRow>& rows) {
  if(rows.empty())return {{},{}};auto next=state;std::vector<double> table;std::vector<std::vector<std::uint32_t>> faults;
  for(const auto& row:rows){const auto result=next.push(row.group,row.timestamp,row.values);table.insert(table.end(),result.first.begin(),result.first.end());faults.push_back(result.second);}
  std::vector<std::uint8_t> missing(table.size(),0);std::vector<double> output(rows.size());
  const auto status=jope_fitted_predict(table.data(),table.size(),missing.data(),missing.size(),rows.size(),state.width,output.data(),output.size());if(status)throw OrderedError("downstream");
  state=std::move(next);return {output,faults};
 }
 OrderedBytes state_bytes()const{return state.to_bytes();}
 void replace_state(const OrderedBytes& bytes){auto next=IntegrationState::from_bytes(bytes,task,state.input,state.cut,state.train_end,state.total_rows);if(next.mode!=state.mode || next.initial.size()!=state.initial.size() || !std::equal(next.initial.begin(),next.initial.end(),state.initial.begin(),[](double a,double b){return ordered_bits(a)==ordered_bits(b);}) || next.width!=state.width || next.ordinal<next.train_end)throw OrderedError("domain");state=std::move(next);}
 void finish()const{state.finish();}
};
