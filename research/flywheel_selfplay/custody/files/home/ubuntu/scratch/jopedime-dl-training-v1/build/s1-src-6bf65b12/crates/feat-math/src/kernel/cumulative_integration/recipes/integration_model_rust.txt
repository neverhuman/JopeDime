#[derive(Clone)]
pub struct OrderedRow {pub group:OrderedGroup,pub timestamp:f64,pub values:Vec<f64>}
pub struct IntegrationModel {state:IntegrationState,predictor:downstream::Predictor,task:u8}
fn fit_ordered(rows:&[OrderedRow],labels:&[f64],task:u8,cut:&str,train_end:usize,total_rows:usize)->Result<IntegrationModel,&'static str> {
    if rows.len()!=train_end || labels.len()!=train_end || labels.iter().any(|v|!v.is_finite()) {return Err("shape");}
    let mut state=IntegrationState::new(@MODE@,@INITIAL@,@WIDTH@,task,@INPUT@,cut,train_end,total_rows)?;
    let mut transformed=Vec::new();for row in rows {transformed.push(state.push(row.group.clone(),row.timestamp,&row.values)?.0);}
    let predictor=downstream::fit(&transformed,labels)?;
    let bytes=state.to_bytes().map_err(|_|"domain")?;
    let array=2+bytes.len().saturating_sub(1)+bytes.iter().map(|v|v.to_string().len()).sum::<usize>();
    let word_array=|values:&[f64]|2+values.len().saturating_sub(1)+values.iter().map(|v|13+v.to_bits().to_string().len()).sum::<usize>();
    let (means,scales,w)=predictor.parameters();
    let fold=format!("{{\"cut_id\":{},\"train_end\":{},\"total_rows\":{}}}",ordered_json_string(cut),train_end,total_rows);
    if @MODEL_BASE_BYTES@+fold.len()+array+word_array(means)+word_array(scales)+word_array(w)>16*1024*1024 {return Err("domain");}
    Ok(IntegrationModel {state,predictor,task})
}
impl IntegrationModel {
    pub fn predict(&mut self,rows:&[OrderedRow])->Result<(Vec<f64>,Vec<Vec<u32>>),&'static str> {
        if rows.is_empty() {return Ok((Vec::new(),Vec::new()));}
        let mut next=self.state.clone();let mut transformed=Vec::new();let mut faults=Vec::new();
        for row in rows {let (values,bits)=next.push(row.group.clone(),row.timestamp,&row.values)?;transformed.push(values);faults.push(bits);}
        let values=self.predictor.predict(&transformed).into_iter().map(|v|if v.is_finite() {v} else {0.0}).collect::<Vec<_>>();if values.len()!=rows.len() {return Err("shape");}self.state=next;Ok((values,faults))
    }
    pub fn state_bytes(&self)->Result<Vec<u8>,&'static str> {self.state.to_bytes()}
    pub fn replace_state(&mut self,bytes:&[u8])->Result<(),&'static str> {
        let next=IntegrationState::from_bytes(bytes,self.task,&self.state.input,&self.state.cut,self.state.train_end,self.state.total_rows)?;
        if next.mode!=self.state.mode || next.initial.iter().map(|v|v.to_bits()).ne(self.state.initial.iter().map(|v|v.to_bits())) || next.width!=self.state.width || next.ordinal<next.train_end {return Err("domain");}
        self.state=next;Ok(())
    }
    pub fn finish(&self)->Result<(),&'static str> {self.state.finish()}
}
