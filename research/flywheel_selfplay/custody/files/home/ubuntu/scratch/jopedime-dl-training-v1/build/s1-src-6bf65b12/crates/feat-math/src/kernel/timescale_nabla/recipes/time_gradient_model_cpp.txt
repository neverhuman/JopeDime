static void time_gradient_checkpoint(const TimeGradientState& state,const Model& predictor) {
    const auto bytes=state.to_bytes();
    const auto& head=predictor.head;
    std::size_t array=2+(bytes.empty()?0:bytes.size()-1);for(auto v:bytes) array+=std::to_string(v).size();
    const auto word_array=[](const std::vector<double>& values) {std::size_t n=2+(values.empty()?0:values.size()-1);for(double v:values) n+=13+std::to_string(ordered_bits(v)).size();return n;};
    const auto fold="{\"cut_id\":"+ordered_json_string(state.cut)+",\"train_end\":"+std::to_string(state.train_end)+",\"total_rows\":"+std::to_string(state.total_rows)+"}";
    if(@MODEL_BASE_BYTES@+fold.size()+array+word_array(head.means)+word_array(head.scales)+word_array(head.w)>16*1024*1024) throw OrderedError("resource");
}
struct OrderedRow {OrderedGroup group;double timestamp;std::vector<double> values;};
struct TimeGradientModel {
    TimeGradientState state;Model predictor;unsigned task;
    std::pair<std::vector<double>,std::vector<std::vector<std::uint32_t>>> predict(const std::vector<OrderedRow>& rows) {
        if(rows.empty()) return {{},{}};auto next=state;std::vector<std::vector<double>> transformed;std::vector<std::vector<std::uint32_t>> faults;
        for(const auto& row:rows) {const auto result=next.push(row.group,row.timestamp,row.values);transformed.push_back(result.first);faults.push_back(result.second);}
        auto values=predictor.predict(transformed);for(auto& value:values) if(!std::isfinite(value)) value=0.0;if(values.size()!=rows.size()) throw OrderedError("shape");time_gradient_checkpoint(next,predictor);state=std::move(next);return {values,faults};
    }
    OrderedBytes state_bytes() const {return state.to_bytes();}
    void replace_state(const OrderedBytes& bytes) {
        auto next=TimeGradientState::from_bytes(bytes,task,state.input,state.cut,state.train_end,state.total_rows);
        if(ordered_bits(next.minimum_gap)!=ordered_bits(state.minimum_gap) || next.width!=state.width || next.ordinal<next.train_end) throw OrderedError("domain");time_gradient_checkpoint(next,predictor);state=std::move(next);
    }
    void finish() const {state.finish();}
};
static TimeGradientModel fit_ordered(const std::vector<OrderedRow>& rows,const std::vector<double>& labels,unsigned task,const std::string& cut,std::size_t train_end,std::size_t total_rows) {
    if(rows.size()!=train_end || labels.size()!=train_end || std::any_of(labels.begin(),labels.end(),[](double v){return !std::isfinite(v);})) throw OrderedError("shape");
    TimeGradientState state(@MINIMUM_GAP@,@WIDTH@,task,@INPUT@,cut,train_end,total_rows);std::vector<std::vector<double>> transformed;
    for(const auto& row:rows) transformed.push_back(state.push(row.group,row.timestamp,row.values).first);
    auto predictor=Model::fit(transformed,labels);
    if(std::all_of(labels.begin(),labels.end(),[](double v){return ordered_bits(v)==ordered_bits(-0.0);})) predictor.head.w[0]=-0.0;
    const auto& head=predictor.head;
    if(head.means.size()!=state.width || head.scales.size()!=state.width || head.w.size()!=state.width+1 || std::any_of(head.means.begin(),head.means.end(),[](double v){return !std::isfinite(v);}) || std::any_of(head.scales.begin(),head.scales.end(),[](double v){return !std::isfinite(v) || v<=0.0;}) || std::any_of(head.w.begin(),head.w.end(),[](double v){return !std::isfinite(v);})) throw OrderedError("domain");
    try {time_gradient_checkpoint(state,predictor);} catch(const OrderedError&) {throw OrderedError("domain");}
    return {std::move(state),std::move(predictor),task};
}
